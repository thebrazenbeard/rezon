# Rezon v0.1 General Multi-Node Solver Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a provider-independent, deterministic multi-node reasoning runtime that preserves the literal proposition, coordinates genuinely different reasoning nodes, structurally opposes the leading candidate, verifies evidence, reconciles conflicts, and emits a canonical result receipt.

**Architecture:** A small Python package carries immutable task/proposition/evidence/claim types through an explicit plan graph. Deterministic workers execute through a node protocol; an opposition stage tries to falsify the leading candidate; reconciliation and verification reject unsupported or contradictory claims; the runtime integrates only surviving claims and serializes a stable receipt.

**Tech Stack:** Python >=3.11, standard library runtime, pytest for tests.

**Spec:** `docs/superpowers/specs/2026-09-15-rezon-v0.1-general-multinode-solver.md`

## Global Constraints

- Python >= 3.11.
- Runtime implementation uses the standard library only.
- `pytest` is the only required development/test dependency.
- Core tests require no network, database, external model, or provider account.
- Deterministic acceptance runs must produce stable canonical JSON bytes for the same inputs.
- Worker failures are typed results and never implicit votes.
- Accepted factual claims require valid evidence references.
- Conflicting supported answers remain `CONFLICT` until one is invalidated by evidence or verification.
- v0.1 source success does not imply merge, package publication, deployment, provider activation, Vera integration, or external effect.

from __future__ import annotations

from dataclasses import asdict, dataclass
from enum import StrEnum
import json
from typing import Any


class Status(StrEnum):
    UNPROVEN = "UNPROVEN"
    COMPLETED = "COMPLETED"
    REJECTED = "REJECTED"
    CONFLICT = "CONFLICT"
    INSUFFICIENT_EVIDENCE = "INSUFFICIENT_EVIDENCE"
    INVALID_TASK = "INVALID_TASK"
    WORKER_FAILED = "WORKER_FAILED"
    RESOURCE_LIMIT = "RESOURCE_LIMIT"


@dataclass(frozen=True)
class TaskEnvelope:
    task_id: str
    literal_request: str
    constraints: tuple[str, ...] = ()
    context_refs: tuple[str, ...] = ()
    resource_budget: int | None = None


@dataclass(frozen=True)
class Proposition:
    proposition_id: str
    text: str
    proposition_type: str
    referent: str | None = None
    scope: str | None = None

    value: Any

# src/rezon/nodes.py
from __future__ import annotations

from typing import Protocol

from .core import Claim, NodeResult, Proposition, Status, TaskEnvelope, WorkerRunReceipt
from .evidence import EvidenceItem


class ReasoningNode(Protocol):
    node_id: str
    operator_type: str
    independent_group: str | None

    def run(
        self,
        task: TaskEnvelope,
        proposition: Proposition,
        evidence: tuple[EvidenceItem, ...],
    ) -> NodeResult: ...


def _contents(evidence: tuple[EvidenceItem, ...]) -> tuple[dict, dict]:
    by_id = {item.evidence_id: item.content for item in evidence}
    return by_id["candidates"], by_id["constraints"]


def _passes(value: float, op: str, threshold: float) -> bool:
    if op == "<=":
        return value <= threshold
    if op == ">=":
        return value >= threshold
    raise ValueError(f"unsupported operator: {op}")


def _feasible_candidates(candidates: dict, constraints: dict) -> list[str]:
    feasible: list[str] = []
    for name in sorted(candidates):
        attrs = candidates[name]
        if all(_passes(attrs[field], rule[0], rule[1]) for field, rule in constraints.items()):
            feasible.append(name)
    return feasible


class ConstraintFilterNode:
    node_id = "constraint_filter"
    operator_type = "deductive_constraint_filter"
    independent_group = "primary"

    def run(self, task, proposition, evidence):
        candidates, constraints = _contents(evidence)
        feasible = _feasible_candidates(candidates, constraints)
        claims = ()
        unresolved = ()
        if len(feasible) == 1:
            claims = (Claim(
                "claim:constraint_filter",
                proposition.proposition_id,
                "Exactly one candidate satisfies every explicit constraint.",
                feasible[0],
                ("candidates", "constraints"),
            ),)
        else:
            unresolved = (f"expected one feasible candidate, found {len(feasible)}",)
        return NodeResult(
            self.node_id,
            claims,
            unresolved,
            None,
            WorkerRunReceipt(self.node_id, Status.COMPLETED, 0),
        )


class IndependentCheckerNode:
    node_id = "independent_checker"
    operator_type = "independent_recomputation"
    independent_group = "crosscheck"

    def run(self, task, proposition, evidence):
        candidates, constraints = _contents(evidence)
        survivors = []
        for name, attrs in sorted(candidates.items()):
            violations = 0
            for field, (op, threshold) in constraints.items():
                if not _passes(attrs[field], op, threshold):
                    violations += 1
            if violations == 0:
                survivors.append(name)
        claims = () if len(survivors) != 1 else (Claim(
            "claim:independent_checker",
            proposition.proposition_id,
            "Independent recomputation found one feasible candidate.",
            survivors[0],
            ("candidates", "constraints"),
        ),)
        return NodeResult(
            self.node_id,
            claims,
            () if claims else ("independent recomputation was not unique",),
            None,
            WorkerRunReceipt(self.node_id, Status.COMPLETED, 0),
        )
```

# src/rezon/planning.py
from dataclasses import dataclass

from .core import Proposition, TaskEnvelope


@dataclass(frozen=True)
class PlanStep:
    node_id: str
    depends_on: tuple[str, ...] = ()


@dataclass(frozen=True)
class PlanGraph:
    plan_id: str
    steps: tuple[PlanStep, ...]

    def __post_init__(self):
        graph = {step.node_id: set(step.depends_on) for step in self.steps}
        visiting: set[str] = set()
        visited: set[str] = set()

        def visit(node: str):
            if node in visiting:
                raise ValueError("plan contains dependency cycle")
            if node in visited:
                return
            visiting.add(node)
            for dep in graph.get(node, ()):
                visit(dep)
            visiting.remove(node)
            visited.add(node)

        for node in graph:
            visit(node)


class Decomposer:
    def build(self, task: TaskEnvelope) -> tuple[Proposition, PlanGraph]:
        if not task.literal_request.strip():
            raise ValueError("literal request must not be empty")
        proposition = Proposition(
            proposition_id=f"prop:{task.task_id}",
            text=task.literal_request,
            proposition_type="constraint_selection",
        )
        plan = PlanGraph(
            plan_id=f"plan:{task.task_id}",
            steps=(
                PlanStep("constraint_filter"),
                PlanStep("independent_checker"),
            ),
        )
        return proposition, plan
```

# src/rezon/execution.py
from dataclasses import replace

from .core import NodeResult, Status, WorkerRunReceipt


class NodeExecutor:
    def __init__(self, nodes: dict[str, object]):
        self._nodes = nodes

    def execute(self, plan, task, proposition, evidence):
        results: list[NodeResult] = []
        completed: set[str] = set()
        remaining = list(plan.steps)
        ordinal = 0
        while remaining:
            ready = [step for step in remaining if set(step.depends_on) <= completed]
            if not ready:
                raise ValueError("plan cannot make progress")
            for step in sorted(ready, key=lambda item: item.node_id):
                node = self._nodes[step.node_id]
                try:
                    result = node.run(task, proposition, evidence)
                    result = replace(
                        result,
                        execution_receipt=replace(result.execution_receipt, ordinal=ordinal),
                    )
                except Exception as exc:
                    result = NodeResult(
                        step.node_id,
                        (),
                        (str(exc),),
                        Status.WORKER_FAILED,
                        WorkerRunReceipt(step.node_id, Status.WORKER_FAILED, ordinal, str(exc)),
                    )
                results.append(result)
                completed.add(step.node_id)
                remaining.remove(step)
                ordinal += 1
        return tuple(results)
```

from dataclasses import dataclass
from .evidence import EvidenceStore


@dataclass(frozen=True)
class VerificationResult:
    accepted: tuple[Claim, ...]
    rejected: tuple[str, ...]


class EvidenceVerifier:
    def verify(self, claims: tuple[Claim, ...], store: EvidenceStore) -> VerificationResult:
        accepted = []
        rejected = []
        for claim in claims:
            if claim.evidence_refs and all(store.contains(ref) for ref in claim.evidence_refs):
                accepted.append(claim)
            else:
                rejected.append(claim.claim_id)
        return VerificationResult(tuple(accepted), tuple(sorted(rejected)))


class OppositionNode:
    node_id = "opposition"
    operator_type = "literal_counterexample_search"
    independent_group = "opposition"

    def challenge(self, candidate: str, evidence: tuple[EvidenceItem, ...]) -> tuple[str, ...]:
        candidates, constraints = _contents(evidence)
        attrs = candidates[candidate]
        counterexamples = []
        for field, (op, threshold) in sorted(constraints.items()):
            if not _passes(attrs[field], op, threshold):
                counterexamples.append(
                    f"{candidate}.{field}={attrs[field]} violates {op} {threshold}"
                )
        return tuple(counterexamples)
```

# src/rezon/reconciliation.py
from dataclasses import dataclass
from typing import Any

from .core import NodeResult, Status


@dataclass(frozen=True)
class IntegrationResult:
    status: Status
    value: Any | None
    accepted_claim_ids: tuple[str, ...]
    conflicts: tuple[str, ...]
    unresolved: tuple[str, ...]


class Reconciler:
    def reconcile(self, results: tuple[NodeResult, ...]) -> IntegrationResult:
        claims = [claim for result in results for claim in result.claims]
        unresolved = tuple(item for result in results for item in result.unresolved)
        if not claims:
            return IntegrationResult(
                Status.INSUFFICIENT_EVIDENCE, None, (), (), unresolved
            )
        by_value: dict[str, list] = {}
        for claim in claims:
            key = repr(claim.value)
            by_value.setdefault(key, []).append(claim)
        if len(by_value) > 1:
            values = tuple(sorted(by_value))
            return IntegrationResult(
                Status.CONFLICT,
                None,
                (),
                values,
                unresolved,
            )
        group = next(iter(by_value.values()))
        return IntegrationResult(
            Status.COMPLETED,
            group[0].value,
            tuple(sorted(claim.claim_id for claim in group)),
            (),
            unresolved,
        )
```

# src/rezon/runtime.py
from .core import ResultReceipt, Status, TaskEnvelope
from .evidence import EvidenceStore
from .execution import NodeExecutor
from .nodes import (
    ConstraintFilterNode,
    EvidenceVerifier,
    IndependentCheckerNode,
    OppositionNode,
)
from .planning import Decomposer
from .reconciliation import Reconciler


class RezonRuntime:
    def __init__(self, decomposer, executor, verifier, opposition, reconciler):
        self.decomposer = decomposer
        self.executor = executor
        self.verifier = verifier
        self.opposition = opposition
        self.reconciler = reconciler

    @classmethod
    def default(cls):
        nodes = {
            "constraint_filter": ConstraintFilterNode(),
            "independent_checker": IndependentCheckerNode(),
        }
        return cls(
            Decomposer(),
            NodeExecutor(nodes),
            EvidenceVerifier(),
            OppositionNode(),
            Reconciler(),
        )

    def solve(self, task: TaskEnvelope, evidence_items):
        store = EvidenceStore(tuple(evidence_items))
        proposition, plan = self.decomposer.build(task)
        raw_results = self.executor.execute(plan, task, proposition, store.all())

        verified_results = []
        rejected = []
        from dataclasses import replace
        for result in raw_results:
            verification = self.verifier.verify(result.claims, store)
            rejected.extend(verification.rejected)
            verified_results.append(replace(result, claims=verification.accepted))

        integration = self.reconciler.reconcile(tuple(verified_results))
        final_status = integration.status
        final_value = integration.value
        conflicts = list(integration.conflicts)

        if final_status == Status.COMPLETED and final_value is not None:
            counterexamples = self.opposition.challenge(final_value, store.all())
            if counterexamples:
                final_status = Status.REJECTED
                conflicts.extend(counterexamples)
                final_value = None

        worker_runs = tuple(
            sorted(
                (result.execution_receipt for result in raw_results),
                key=lambda receipt: (receipt.ordinal, receipt.node_id),
            )
        )
        return ResultReceipt(
            task_id=task.task_id,
            plan_id=plan.plan_id,
            proposition_id=proposition.proposition_id,
            worker_runs=worker_runs,
            accepted_claim_ids=tuple(sorted(integration.accepted_claim_ids)),
            rejected_claim_ids=tuple(sorted(rejected)),
            conflicts=tuple(sorted(conflicts)),
            unresolved=tuple(sorted(integration.unresolved)),
            final_status=final_status,
            final_value=final_value,
        )
```
